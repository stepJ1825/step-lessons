-- 26. Самый загруженный день недели в Европе
-- (упрощённо: по кодам крупных аэропортов)
SELECT
    EXTRACT(ISODOW FROM f.scheduled_departure) AS dow,
    COUNT(*) AS flights
FROM flights f
join routes r on r.route_no = f.route_no
WHERE departure_airport IN ('LHR', 'CDG', 'FRA', 'AMS', 'MAD', 'FCO', 'IST')
GROUP BY dow
ORDER BY flights DESC
LIMIT 1;
--EXTRACT(... FROM ...) — извлекает часть даты/времени.
--ISODOW — ISO день недели (от 1 до 7)

-- 27. Недоиспользованные рейсы (<25% мест, неделя)
WITH capacity AS (
    SELECT f.flight_id, COUNT(s.seat_no) AS total
    FROM flights f
    join routes r on f.route_no = f.route_no 
    join airplanes_data ad on ad.airplane_code =r.airplane_code 
    JOIN seats s ON ad.airplane_code = s.airplane_code 
    WHERE f.scheduled_departure >= CURRENT_DATE - INTERVAL '7 days'
    	and f.scheduled_departure <= CURRENT_DATE
    GROUP BY f.flight_id
),
occupied AS (
    SELECT flight_id, COUNT(*) AS sold
    FROM boarding_passes
    GROUP BY flight_id
)
SELECT
    c.flight_id,
    c.total,
    COALESCE(o.sold, 0) AS sold,
    COALESCE(o.sold, 0)::float / c.total AS occupancy
FROM capacity c
LEFT JOIN occupied o ON c.flight_id = o.flight_id
WHERE COALESCE(o.sold, 0)::float / c.total < 0.25
ORDER BY occupancy DESC;

-- 28. Прилёт раньше вылета (по UTC)
SELECT flight_id, scheduled_departure, scheduled_arrival
FROM flights
WHERE scheduled_departure::time > scheduled_arrival::time
and scheduled_departure::date <= scheduled_arrival::date;


-- 29. Пассажиры >5 полётов за год
SELECT passenger_name, COUNT(*) AS flights
FROM tickets t
JOIN boarding_passes bp ON t.ticket_no = bp.ticket_no
JOIN flights f ON bp.flight_id = f.flight_id
WHERE f.scheduled_departure >= CURRENT_DATE - INTERVAL '1 year'
GROUP BY passenger_name
HAVING COUNT(*) > 5
ORDER BY flights DESC;
